: 20, background: active === key ? "#1C3F6E" : "transparent", display: "flex", flexDirection: "column", alignItems: "center", gap: 3, transition: "background 0.15s" }}>
            <Icon active={active === key} />
            <span style={{ ...T.tabLbl, color: active === key ? "#fff" : "#9B9B9B", fontFamily: FF }}>{label}</span>
          </div>
        </button>
      ))}
    </div>
  );
};
